# Counterpart and verifier dependencies. The core tier needs none of these - it is stdlib-only - so a
# laptop run of `ci/e2e/run.py --tier core` works with a bare python3. CodeBuild installs all of them.

# AWS destinations: verifiers (S3, SQS, Kinesis, SiteWise, ...) and run-scoped cleanup.
boto3>=1.35,<2

# S3 Tables read-back through the Iceberg REST catalog with SigV4 (signing name s3tables). Metadata alone
# cannot attribute rows to a run, so the verifier filters rows by the run marker.
pyiceberg[pyarrow]>=0.9,<0.11

# OPC-UA counterparts: a writable server for opcua-writer-target (SFC's own opcua-target serves
# READ_ONLY nodes, OpcuaNamespaceBuilder.kt:128) and an independent client probe.
asyncua>=1.1.5,<2

# MQTT and NATS subscribers that read back what the mqtt and nats targets published.
paho-mqtt>=2.1,<3
nats-py>=2.7,<3

# Modbus-TCP server for the modbus-tcp adapter. Pinned: the server/datastore API changes across 3.x.
pymodbus==3.6.9
